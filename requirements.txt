# Python dependencies for the compute backends and the warehouse connection.
#
# Pinned, not floated. Two of these have already broken each other in this repo:
# `flightsql-dbapi` pins `sqlalchemy<2`, and installing it silently downgraded
# the sqlalchemy that PyIceberg's SqlCatalog needs — so the Iceberg conformance
# leg started failing for a reason that had nothing to do with Iceberg. A range
# would let that recur on a fresh machine and present as a compiler bug.
#
#   python3 -m pip install -r requirements.txt
#
# Nothing here is needed to run the authoring surface, which evaluates rules in
# the browser. These are for `npm run verify` (the conformance harness) and for
# reading a real warehouse.

# --- conformance: the reference engine and the two compute targets ----------
duckdb==1.5.5
polars==1.43.1
pyarrow==25.0.0

# --- Iceberg: catalogue, snapshot reads, partition-scoped writes ------------
pyiceberg==0.11.1
# PyIceberg's SqlCatalog is a SQLAlchemy application. Held at 2.x deliberately:
# anything that drags in `flightsql-dbapi` will try to pull this back to 1.4.
sqlalchemy==2.0.51

# --- Dremio: Arrow Flight SQL over gRPC ------------------------------------
# ADBC rather than `flightsql-dbapi` — same protocol, no sqlalchemy pin, and the
# Arrow batches arrive without a DB-API layer in between.
adbc-driver-flightsql==1.12.0
adbc-driver-manager==1.12.0
